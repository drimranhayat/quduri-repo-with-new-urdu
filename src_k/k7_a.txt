@kitab 7 | كِتَابُ الرَّهْنِ ### @kitab 7 | کتاب الرہن
@bab | انْعِقَادُ الرَّهْنِ وَضَمَانُهُ ### @bab | رہن کا انعقاد اور اس کا ضمان
~ الِانْعِقَادُ وَالْقَبْضُ ### ~ انعقاد اور قبضہ
الرَّهْنُ: يَنْعَقِدُ بِالْإِيجَابِ وَالْقَبُولِ وَيَتِمُّ بِالْقَبْضِ ### رہن ایجاب و قبول سے منعقد ہوتا ہے اور قبضے سے مکمل ہوتا ہے
فَإِذَا قَبَضَ الْمُرْتَهِنُ الرَّهْنَ مَحُوزًا مُفَرَّغًا مُمَيَّزًا تَمَّ الْعَقْدُ فِيهِ ### جب مرتہن نے رہن کو اس حال میں قبضے میں لیا کہ وہ الگ، خالی اور ممتاز تھا تو اس میں عقد مکمل ہو گیا
وَمَا لَمْ يَقْبِضْهُ فَالرَّاهِنُ بِالْخِيَارِ: إِنْ شَاءَ سَلَّمَهُ وَإِنْ شَاءَ رَجَعَ عَنِ الرَّهْنِ ### اور جب تک مرتہن نے قبضہ نہیں کیا راہن کو اختیار ہے: چاہے حوالے کر دے اور چاہے رہن سے رجوع کر لے
فَإِذَا سَلَّمَهُ إِلَيْهِ وَقَبَضَهُ دَخَلَ فِي ضَمَانِهِ ### جب راہن نے حوالے کیا اور مرتہن نے قبضہ کر لیا تو رہن اس کے ضمان میں آ گیا
~ مِقْدَارُ الضَّمَانِ ### ~ ضمان کی مقدار
وَلَا يَصِحُّ الرَّهْنُ إِلَّا بِدَيْنٍ مَضْمُونٍ ### رہن صرف مضمون دین کے بدلے صحیح ہے
وَهُوَ مَضْمُونٌ بِالْأَقَلِّ مِنْ قِيمَتِهِ وَمِنَ الدَّيْنِ ### اور رہن قیمت اور دین میں سے کم کے برابر مضمون ہوتا ہے
فَإِذَا هَلَكَ فِي يَدِ الْمُرْتَهِنِ وَقِيمَتُهُ وَالدَّيْنُ سَوَاءٌ صَارَ الْمُرْتَهِنُ مُسْتَوْفِيًا لِدَيْنِهِ حُكْمًا ### اگر مرتہن کے ہاتھ میں ہلاک ہو گیا اور اس کی قیمت اور دین برابر ہوں تو مرتہن حکماً اپنا دین وصول کر چکا مانا جائے گا
وَإِنْ كَانَتْ قِيمَةُ الرَّهْنِ أَكْثَرَ مِنَ الدَّيْنِ فَالْفَضْلُ أَمَانَةٌ فِي يَدِهِ ### اگر رہن کی قیمت دین سے زیادہ ہو تو زائد حصہ اس کے ہاتھ میں امانت ہے
وَإِنْ كَانَتْ أَقَلَّ سَقَطَ مِنَ الدَّيْنِ بِقَدْرِهَا وَرَجَعَ الْمُرْتَهِنُ بِالْفَضْلِ ### اور اگر کم ہو تو دین اتنا ساقط ہو جائے گا اور مرتہن بقیہ کے لیے رجوع کرے گا
~ مَا يَصِحُّ رَهْنُهُ وَمَا لَا يَصِحُّ ### ~ کیا رہن رکھنا صحیح ہے اور کیا نہیں
وَلَا يَجُوزُ رَهْنُ الْمُشَاعِ ### مشترک (غیر منقسم) چیز کا رہن جائز نہیں
وَلَا رَهْنُ ثَمَرَةٍ عَلَى رُؤُوسِ النَّخْلِ دُونَ النَّخْلِ وَلَا زَرْعٍ فِي الْأَرْضِ دُونَ الْأَرْضِ ### نہ درخت کے بغیر درخت پر لگے پھل کا رہن اور نہ زمین کے بغیر زمین میں کھڑی کھیتی کا
وَلَا يَجُوزُ رَهْنُ الْأَرْضِ وَالنَّخْلِ دُونَهُمَا ### اور زمین و درخت کا ان (کھیتی و پھل) کے بغیر رہن بھی جائز نہیں
وَلَا يَصِحُّ الرَّهْنُ بِالْأَمَانَاتِ كَالْوَدَائِعِ وَالْمُضَارَبَاتِ وَمَالِ الشَّرِكَةِ ### امانتوں کے بدلے رہن صحیح نہیں، جیسے ودیعتیں، مضاربت اور شرکت کا مال
وَيَصِحُّ الرَّهْنُ بِرَأْسِ مَالِ السَّلَمِ وَثَمَنِ الصَّرْفِ وَالْمُسْلَمِ فِيهِ ### سلم کے رأس المال، صرف کے ثمن اور مسلم فیہ کے بدلے رہن صحیح ہے
فَإِنْ هَلَكَ فِي مَجْلِسِ الْعَقْدِ تَمَّ الصَّرْفُ وَالسَّلَمُ وَصَارَ الْمُرْتَهِنُ مُسْتَوْفِيًا لِدَيْنِهِ ### اگر مجلسِ عقد میں رہن ہلاک ہو گیا تو صرف اور سلم مکمل ہو گئے اور مرتہن اپنا دین وصول کر چکا مانا جائے گا
~ الْعَدْلُ وَرَهْنُ النُّقُودِ ### ~ عادل کے پاس رہن اور نقد کا رہن
وَإِذَا اتَّفَقَا عَلَى وَضْعِ الرَّهْنِ عَلَى يَدِ عَدْلٍ جَازَ ### اگر دونوں رہن کو کسی عادل کے ہاتھ میں رکھنے پر متفق ہو جائیں تو جائز ہے
وَلَيْسَ لِلْمُرْتَهِنِ وَلَا لِلرَّاهِنِ أَخْذُهُ مِنْ يَدِهِ ### اور نہ مرتہن کو حق ہے اور نہ راہن کو کہ اسے عادل کے ہاتھ سے لے لے
فَإِنْ هَلَكَ فِي يَدِهِ هَلَكَ مِنْ ضَمَانِ الْمُرْتَهِنِ ### اگر عادل کے ہاتھ میں ہلاک ہو گیا تو مرتہن کے ضمان سے ہلاک ہوا
وَيَجُوزُ رَهْنُ الدَّرَاهِمِ وَالدَّنَانِيرِ وَالْمَكِيلِ وَالْمَوْزُونِ ### درہم، دینار، ناپی اور تولی جانے والی چیزوں کا رہن جائز ہے
فَإِنْ رُهِنَتْ بِجِنْسِهَا وَهَلَكَتْ هَلَكَتْ بِمِثْلِهَا مِنَ الدَّيْنِ وَإِنِ اخْتَلَفَا فِي الْجَوْدَةِ وَالصِّنَاعَةِ ### اگر اپنی ہی جنس کے دین کے بدلے رہن رکھے گئے اور ہلاک ہو گئے تو دین کی اتنی مثل کے بدلے ہلاک ہوئے اگرچہ معیار اور ساخت میں اختلاف ہو
وَمَنْ كَانَ لَهُ دَيْنٌ عَلَى غَيْرِهِ فَأَخَذَ مِنْهُ مِثْلَ دَيْنِهِ فَأَنْفَقَهُ ثُمَّ عَلِمَ أَنَّهُ كَانَ زُيُوفًا فَلَا شَيْءَ لَهُ عِنْدَ أَبِي حَنِيفَةَ ### جس کا کسی پر دین تھا، پھر اس نے اپنے دین کے برابر رقم لے کر خرچ کر لی، پھر معلوم ہوا کہ وہ کھوٹے سکے تھے تو امام ابوحنیفہؒ کے نزدیک اسے کچھ نہیں ملے گا
وَقَالَ أَبُو يُوسُفَ وَمُحَمَّدٌ: يَرُدُّ مِثْلَ الزُّيُوفِ وَيَرْجِعُ بِالْجِيَادِ ### امام ابو یوسفؒ و امام محمدؒ نے فرمایا: کھوٹوں کی مثل لوٹائے گا اور کھرے سکے وصول کرے گا
وَمَنْ رَهَنَ عَبْدَيْنِ بِأَلْفِ دِرْهَمٍ فَقَضَى حِصَّةَ أَحَدِهِمَا لَمْ يَكُنْ لَهُ أَنْ يَسْتَرِدَّهُ حَتَّى يُؤَدِّيَ بَاقِيَ الدَّيْنِ ### جس نے دو غلام ہزار درہم کے بدلے رہن رکھے، پھر ایک کے حصے کا دین ادا کر دیا تو اسے حق نہیں کہ اسے واپس لے جب تک باقی دین ادا نہ کر دے
@bab | تَصَرُّفَاتٌ فِي الرَّهْنِ وَأَحْكَامُهُ ### @bab | رہن میں تصرفات اور اس کے احکام
~ بَيْعُ الرَّهْنِ وَالتَّوْكِيلُ بِهِ ### ~ رہن کی بیع اور اس کی وکالت
وَإِذَا وَكَّلَ الرَّاهِنُ الْمُرْتَهِنَ أَوِ الْعَدْلَ أَوْ غَيْرَهُمَا بِبَيْعِ الرَّهْنِ عِنْدَ حُلُولِ الدَّيْنِ فَالْوَكَالَةُ جَائِزَةٌ ### اگر راہن نے مرتہن، عادل یا کسی اور کو دین کی میعاد آنے پر رہن بیچنے کا وکیل بنایا تو وکالت جائز ہے
فَإِنْ شُرِطَتْ فِي عَقْدِ الرَّهْنِ فَلَيْسَ لِلرَّاهِنِ عَزْلُهُ عَنْهَا فَإِنْ عَزَلَهُ لَمْ يَنْعَزِلْ ### اگر یہ وکالت عقدِ رہن میں شرط کی گئی تو راہن اسے معزول نہیں کر سکتا، معزول کرے تو بھی معزول نہ ہو گا
وَإِنْ مَاتَ الرَّاهِنُ لَمْ يَنْعَزِلْ ### اور اگر راہن مر جائے تو بھی وہ معزول نہ ہو گا
وَلِلْمُرْتَهِنِ أَنْ يُطَالِبَ الرَّاهِنَ وَيَحْبِسَهُ بِهِ ### مرتہن راہن سے دین کا مطالبہ کر سکتا ہے اور اسے اس کی وجہ سے قید کروا سکتا ہے
وَإِنْ كَانَ الرَّهْنُ فِي يَدِهِ فَلَيْسَ عَلَيْهِ أَنْ يُمَكِّنَهُ مِنْ بَيْعِهِ حَتَّى يَقْضِيَهُ الدَّيْنَ مِنْ ثَمَنِهِ ### اگر رہن اس کے ہاتھ میں ہو تو اس پر لازم نہیں کہ راہن کو بیچنے دے جب تک وہ اس کے ثمن سے دین ادا نہ کر دے
فَإِنْ قَضَاهُ الدَّيْنَ قِيلَ لَهُ: سَلِّمِ الرَّهْنَ إِلَيْهِ ### اگر راہن نے دین ادا کر دیا تو مرتہن سے کہا جائے گا: رہن اس کے حوالے کرو
وَإِذَا بَاعَ الرَّاهِنُ الرَّهْنَ بِغَيْرِ إِذْنِ الْمُرْتَهِنِ فَالْبَيْعُ مَوْقُوفٌ ### اگر راہن نے مرتہن کی اجازت کے بغیر رہن بیچ دیا تو بیع موقوف ہے
فَإِنْ أَجَازَهُ الْمُرْتَهِنُ جَازَ وَإِنْ قَضَاهُ الرَّاهِنُ دَيْنَهُ جَازَ الْبَيْعُ ### اگر مرتہن نے اجازت دے دی تو بیع جائز ہے، اور اگر راہن نے مرتہن کا دین ادا کر دیا تب بھی بیع جائز ہے
~ عِتْقُ الرَّهْنِ وَالْجِنَايَةُ عَلَيْهِ ### ~ رہن کی آزادی اور اس پر جنایت
وَإِنْ أَعْتَقَ الرَّاهِنُ عَبْدَ الرَّهْنِ نَفَذَ عِتْقُهُ ### اگر راہن نے رہن کے غلام کو آزاد کر دیا تو اس کی آزادی نافذ ہو گئی
فَإِنْ كَانَ الدَّيْنُ حَالًّا طُولِبَ بِأَدَاءِ الدَّيْنِ ### اگر دین نقد (واجب الادا) ہو تو راہن سے دین ادا کرنے کا مطالبہ کیا جائے گا
وَإِنْ كَانَ مُؤَجَّلًا أُخِذَ مِنْهُ قِيمَةُ الْعَبْدِ فَجُعِلَتْ رَهْنًا مَكَانَهُ حَتَّى يَحِلَّ الدَّيْنُ ### اور اگر ادھار ہو تو اس سے غلام کی قیمت لے کر اس کی جگہ رہن بنا دی جائے گی یہاں تک کہ دین کی میعاد آ جائے
وَإِنْ كَانَ الرَّاهِنُ مُعْسِرًا اسْتُسْعِيَ الْعَبْدُ فِي قِيمَتِهِ فَقَضَى بِهَا دَيْنَهُ ### اور اگر راہن تنگ دست ہو تو غلام سے اس کی قیمت کی کمائی کروائی جائے گی اور اس سے دین ادا ہو گا
وَكَذَلِكَ إِذَا اسْتَهْلَكَ الرَّاهِنُ الرَّهْنَ ### اسی طرح جب راہن رہن کو ہلاک کر دے
وَإِنِ اسْتَهْلَكَهُ أَجْنَبِيٌّ فَالْمُرْتَهِنُ هُوَ الْخَصْمُ فِي تَضْمِينِهِ وَيَأْخُذُ الْقِيمَةَ فَتَكُونُ رَهْنًا فِي يَدِهِ ### اگر اجنبی نے ہلاک کیا تو اس سے ضمان لینے میں مرتہن فریق ہے؛ وہ قیمت لے گا اور وہ اس کے ہاتھ میں رہن رہے گی
وَجِنَايَةُ الرَّاهِنِ عَلَى الرَّهْنِ مَضْمُونَةٌ ### راہن کی رہن پر جنایت مضمون ہے
وَجِنَايَةُ الْمُرْتَهِنِ عَلَيْهِ تَسْقُطُ مِنْ دَيْنِهِ بِقَدْرِهَا ### اور مرتہن کی رہن پر جنایت سے اتنی مقدار اس کے دین سے ساقط ہو جاتی ہے
وَجِنَايَةُ الرَّهْنِ عَلَى الرَّاهِنِ وَعَلَى الْمُرْتَهِنِ وَعَلَى مَالِهِمَا هَدَرٌ ### رہن کی راہن، مرتہن اور ان کے مال پر جنایت رائیگاں ہے
~ نَفَقَةُ الرَّهْنِ وَنَمَاؤُهُ ### ~ رہن کا خرچ اور اس کا نفع
وَأُجْرَةُ الْبَيْتِ الَّذِي يُحْفَظُ فِيهِ الرَّهْنُ عَلَى الْمُرْتَهِنِ ### جس گھر میں رہن محفوظ رکھا جائے اس کا کرایہ مرتہن پر ہے
وَأُجْرَةُ الرَّاعِي وَنَفَقَةُ الرَّهْنِ عَلَى الرَّاهِنِ وَنَمَاؤُهُ لِلرَّاهِنِ فَيَكُونُ رَهْنًا مَعَ الْأَصْلِ ### چرواہے کی اجرت اور رہن کا خرچ راہن پر ہے، اور اس کا نفع راہن کا ہے اور وہ اصل کے ساتھ رہن ہو گا
فَإِنْ هَلَكَ النَّمَاءُ هَلَكَ بِغَيْرِ شَيْءٍ ### اگر نفع ہلاک ہو جائے تو بغیر کسی عوض کے ہلاک ہوا
وَإِنْ هَلَكَ الْأَصْلُ وَبَقِيَ النَّمَاءُ افْتَكَّهُ الرَّاهِنُ بِحِصَّتِهِ ### اور اگر اصل ہلاک ہو جائے اور نفع باقی رہے تو راہن اسے اس کے حصے کے بدلے چھڑا لے گا
وَيُقْسَمُ الدَّيْنُ عَلَى قِيمَةِ الرَّهْنِ يَوْمَ الْقَبْضِ وَقِيمَةِ النَّمَاءِ يَوْمَ الْفِكَاكِ ### اور دین رہن کی قبضے کے دن کی قیمت اور نفع کی چھڑانے کے دن کی قیمت پر تقسیم ہو گا
فَمَا أَصَابَ الْأَصْلَ سَقَطَ مِنَ الدَّيْنِ وَمَا أَصَابَ النَّمَاءَ افْتَكَّهُ الرَّاهِنُ بِهِ ### جو حصہ اصل کو پہنچے وہ دین سے ساقط ہو گا اور جو نفع کو پہنچے اس کے بدلے راہن نفع کو چھڑا لے گا
وَتَجُوزُ الزِّيَادَةُ فِي الرَّهْنِ وَلَا تَجُوزُ فِي الدَّيْنِ عِنْدَ أَبِي حَنِيفَةَ وَمُحَمَّدٍ وَلَا يَصِيرُ الرَّهْنُ رَهْنًا بِهِمَا ### رہن میں اضافہ جائز ہے اور امام ابوحنیفہؒ و امام محمدؒ کے نزدیک دین میں اضافہ جائز نہیں، اور رہن دونوں (اصل و اضافے) کے بدلے رہن نہیں ہوتا
وَقَالَ أَبُو يُوسُفَ: تَجُوزُ الزِّيَادَةُ فِي الدَّيْنِ أَيْضًا ### امام ابو یوسفؒ نے فرمایا: دین میں اضافہ بھی جائز ہے
@bab | مَسَائِلُ مُتَفَرِّقَةٌ فِي الرَّهْنِ ### @bab | رہن کے متفرق مسائل
وَإِذَا رَهَنَ عَيْنًا وَاحِدَةً عِنْدَ رَجُلَيْنِ بِدَيْنٍ لِكُلِّ وَاحِدٍ مِنْهُمَا جَازَ ### اگر ایک ہی چیز دو آدمیوں کے پاس رہن رکھی، ہر ایک کے دین کے بدلے، تو جائز ہے
وَجَمِيعُهَا رَهْنٌ عِنْدَ كُلِّ وَاحِدٍ مِنْهُمَا وَالْمَضْمُونُ عَلَى كُلِّ وَاحِدٍ مِنْهُمَا حِصَّةُ دَيْنِهِ مِنْهَا ### پوری چیز ہر ایک کے پاس رہن ہے اور ہر ایک پر اس کے دین کے حصے کے بقدر ضمان ہے
فَإِنْ قَضَى أَحَدَهُمَا دَيْنَهُ كَانَتْ كُلُّهَا رَهْنًا فِي يَدِ الْآخَرِ حَتَّى يَسْتَوْفِيَ دَيْنَهُ ### اگر ایک کا دین ادا کر دیا تو پوری چیز دوسرے کے پاس رہن رہے گی یہاں تک کہ وہ اپنا دین وصول کر لے
وَمَنْ بَاعَ عَبْدًا عَلَى أَنْ يَرْهَنَهُ الْمُشْتَرِي بِالثَّمَنِ شَيْئًا بِعَيْنِهِ فَامْتَنَعَ الْمُشْتَرِي مِنْ تَسْلِيمِ الرَّهْنِ لَمْ يُجْبَرْ عَلَيْهِ ### جس نے غلام اس شرط پر بیچا کہ خریدار ثمن کے بدلے کوئی معین چیز رہن رکھے گا، پھر خریدار نے رہن دینے سے انکار کیا تو اسے مجبور نہیں کیا جائے گا
وَكَانَ الْبَائِعُ بِالْخِيَارِ: إِنْ شَاءَ رَضِيَ بِتَرْكِ الرَّهْنِ وَإِنْ شَاءَ فَسَخَ الْبَيْعَ ### اور بیچنے والے کو اختیار ہو گا: چاہے رہن کے ترک پر راضی ہو جائے اور چاہے بیع فسخ کر دے
إِلَّا أَنْ يَدْفَعَ الْمُشْتَرِي الثَّمَنَ حَالًّا أَوْ يَدْفَعَ قِيمَةَ الرَّهْنِ رَهْنًا مَكَانَهُ ### الا یہ کہ خریدار ثمن نقد ادا کر دے یا رہن کی قیمت اس کی جگہ رہن کے طور پر دے دے
وَلِلْمُرْتَهِنِ أَنْ يَحْفَظَ الرَّهْنَ بِنَفْسِهِ وَزَوْجَتِهِ وَوَلَدِهِ وَخَادِمِهِ الَّذِي فِي عِيَالِهِ ### مرتہن رہن کی حفاظت خود، اپنی بیوی، اولاد اور اپنے زیرِ کفالت خادم سے کروا سکتا ہے
وَإِنْ حَفِظَهُ بِغَيْرِ مَنْ فِي عِيَالِهِ أَوْ أَوْدَعَهُ ضَمِنَ ### اور اگر اپنے زیرِ کفالت لوگوں کے علاوہ سے حفاظت کروائی یا کسی کو امانت رکھوا دیا تو ضامن ہو گا
وَإِذَا تَعَدَّى الْمُرْتَهِنُ فِي الرَّهْنِ ضَمِنَهُ ضَمَانَ الْغَصْبِ بِجَمِيعِ قِيمَتِهِ ### جب مرتہن رہن میں زیادتی کرے تو غصب کے ضمان کی طرح پوری قیمت کا ضامن ہو گا
وَإِذَا أَعَارَ الْمُرْتَهِنُ الرَّهْنَ لِلرَّاهِنِ فَقَبَضَهُ خَرَجَ مِنْ ضَمَانِ الْمُرْتَهِنِ ### اگر مرتہن نے رہن راہن کو عاریت پر دیا اور راہن نے قبضہ کر لیا تو وہ مرتہن کے ضمان سے نکل گیا
فَإِنْ هَلَكَ فِي يَدِ الرَّاهِنِ هَلَكَ بِغَيْرِ شَيْءٍ ### پھر اگر راہن کے ہاتھ میں ہلاک ہو جائے تو بغیر کسی عوض کے ہلاک ہوا
وَلِلْمُرْتَهِنِ أَنْ يَسْتَرْجِعَهُ إِلَى يَدِهِ فَإِذَا أَخَذَهُ عَادَ الضَّمَانُ ### مرتہن کو حق ہے کہ اسے اپنے ہاتھ میں واپس لے لے، جب لے لے تو ضمان لوٹ آتا ہے
وَإِذَا مَاتَ الرَّاهِنُ بَاعَ وَصِيُّهُ الرَّهْنَ وَقَضَى الدَّيْنَ ### راہن مر جائے تو اس کا وصی رہن بیچے گا اور دین ادا کرے گا
فَإِنْ لَمْ يَكُنْ لَهُ وَصِيٌّ نَصَبَ الْقَاضِي لَهُ وَصِيًّا وَأَمَرَهُ بِبَيْعِهِ ### اگر اس کا کوئی وصی نہ ہو تو قاضی اس کے لیے وصی مقرر کرے گا اور اسے بیچنے کا حکم دے گا
